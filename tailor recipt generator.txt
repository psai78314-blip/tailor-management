<script>

function generateReceipt()
{
    let customerName =
    document.getElementById("customerName").value;

    let dressType =
    document.getElementById("dressType").value;

    let measurement =
    document.getElementById("measurement").value;

    let amount =
    document.getElementById("amount").value;

    if(customerName == "" ||
       dressType == "" ||
       measurement == "" ||
       amount == "")
    {
        alert("Please fill all fields");
        return;
    }

    // RANDOM RECEIPT NUMBER

    let receiptNo =
    Math.floor(Math.random() * 100000);

    // DISPLAY RECEIPT

    document.getElementById("receipt")
    .style.display = "block";

    document.getElementById("rno")
    .innerHTML = receiptNo;

    document.getElementById("rname")
    .innerHTML = customerName;

    document.getElementById("rdress")
    .innerHTML = dressType;

    document.getElementById("rmeasurement")
    .innerHTML = measurement;

    document.getElementById("ramount")
    .innerHTML = amount;
}

function printReceipt()
{
    window.print();
}
